Stage 3 in a nutshelllines 1–23

Reads results.json and nothing else, so figures can be restyled and redrawn in seconds (stages=[visualize]) without recomputing anything. All numbers are shown in percent as mean ± std over the repeat seeds. matplotlib.use("Agg") draws to files without needing a screen (important on servers).

best_rows(): one number per (d, K)lines 29–36

A cell may have been evaluated at several anchor budgets. For the summary figures each predictor gets its best budget, judged by mean full accuracy.

29def best_rows(rows, model):
30    """For one model, the row with the best mean full accuracy in each (d, K) cell."""
31    best = {}
32    for r in rows:
33        key = (r["d"], r["K"])
34        if r["model"] == model and (key not in best or r["full_acc"] > best[key]["full_acc"]):
35            best[key] = r
36    return best

plot_model(): the heatmapslines 39–61

Three panels (full accuracy, mode F1, hallucination F1). Dimension d runs left to right, the number of modes K top to bottom, colour = mean score from 0 to 1, and each square is labelled "mean±std" in percent. Empty cells stay blank; the text turns white on dark squares.

39def plot_model(cells, ds, Ks, model, title, out_dir):
40    """One row of heatmaps (d across, K down), coloured by the mean, labelled 'mean+-std'."""
41    fig, axes = plt.subplots(1, len(PANELS), figsize=(4.2 * len(PANELS), 0.55 * len(Ks) + 3.0), squeeze=False)
42    for ax, (key, label) in zip(axes[0], PANELS):
43        mean = np.array([[cells[(d, K)][key] if (d, K) in cells else NAN for d in ds] for K in Ks])
44        std = np.array([[cells[(d, K)].get(key + "_std", NAN) if (d, K) in cells else NAN for d in ds] for K in Ks])
45        im = ax.imshow(mean, vmin=0, vmax=1, aspect="auto", cmap="viridis")
46        ax.set_xticks(range(len(ds))); ax.set_xticklabels(ds)
47        ax.set_yticks(range(len(Ks))); ax.set_yticklabels(Ks)
48        ax.set_xlabel("dimension $d$"); ax.set_ylabel("modes $K$")
49        ax.set_title(label, fontsize=10)
50        for i in range(len(Ks)):
51            for j in range(len(ds)):
52                v, sd = mean[i, j], std[i, j]
53                if v == v:                                       # skip empty cells (NaN)
54                    txt = f"{v * 100:.0f}" if sd != sd else f"{v * 100:.0f}$\\pm${sd * 100:.0f}"
55                    ax.text(j, i, txt, ha="center", va="center", fontsize=7,
56                            color="white" if v < 0.6 else "black")
57    fig.colorbar(im, ax=axes[0].tolist(), fraction=0.025, pad=0.02)
58    fig.suptitle(title, fontsize=12)
59    path = os.path.join(out_dir, f"{model}.png")
60    fig.savefig(path, dpi=150, bbox_inches="tight"); plt.close(fig)
61    return path

table_rows()lines 64–74

The table as (K, d, [(mean, std) per predictor]) rows, at each predictor's best budget.

64def table_rows(rows, ds, Ks, models):
65    """[(K, d, [(mean, std) of full accuracy per model])], at each model's best budget."""
66    out = []
67    for K in Ks:
68        for d in ds:
69            vals = []
70            for m in models:
71                r = best_rows(rows, m).get((d, K))
72                vals.append((r["full_acc"], r.get("full_acc_std", NAN)) if r else (NAN, NAN))
73            out.append((K, d, vals))
74    return out

write_tex(): a table ready for the paperlines 77–96

A LaTeX table using the booktabs package, one block of rows per K separated by a rule, "--" where there is no result.

77def write_tex(rows, ds, Ks, models, name, T_true, n_seeds, out_dir):
78    lines = ["% Auto-generated by visualize.py -- requires \\usepackage{booktabs}",
79             "\\begin{table}[t]", "\\centering",
80             f"\\caption{{Seed-fate full accuracy (\\%, mean $\\pm$ std over {n_seeds} seeds) on the "
81             f"$(d,K)$ grid ({name}, $T={T_true}$), best anchor budget per cell.}}",
82             f"\\label{{tab:atlas-{name}}}",
83             f"\\begin{{tabular}}{{{'cc ' + 'r' * len(models)}}}", "\\toprule",
84             "$K$ & $d$ & " + " & ".join(models) + " \\\\", "\\midrule"]
85    prev_K = None
86    for K, d, vals in table_rows(rows, ds, Ks, models):
87        if prev_K is not None and K != prev_K:
88            lines.append("\\midrule")                            # a rule between K blocks
89        prev_K = K
90        cells = " & ".join("--" if mu != mu else f"{mu * 100:.1f} $\\pm$ {sd * 100:.1f}" for mu, sd in vals)
91        lines.append(f"{K} & {d} & {cells} \\\\")
92    lines += ["\\bottomrule", "\\end{tabular}", "\\end{table}"]
93    path = os.path.join(out_dir, "table.tex")
94    with open(path, "w") as f:
95        f.write("\n".join(lines) + "\n")
96    return path

write_png(): the same table as an imagelines 99–118

Bold header, striped rows; for a quick look without LaTeX.

 99def write_png(rows, ds, Ks, models, name, T_true, n_seeds, out_dir):
100    fmt = lambda mu, sd: "--" if mu != mu else f"{mu * 100:.1f}$\\pm${sd * 100:.1f}"
101    header = ["$K$", "$d$"] + models

103             for K, d, vals in table_rows(rows, ds, Ks, models)]

107    tb.auto_set_font_size(False); tb.set_fontsize(8); tb.scale(1, 1.35)

115                 f"best budget", fontsize=10, pad=12)
116    path = os.path.join(out_dir, "table.png")

run(): draw everythinglines 121–162

Finds this run's results.json (never another run's), then writes the table (tex + png), one heatmap figure per predictor at its best budget, and one per predictor for each budget in anchors_<b>/. Each figure is attempted separately, so one failing figure is reported without stopping the rest.

121def run(cfg):
122    out_dir = resolve_sweep_dir(cfg.paths.output, cfg.run_id, cfg.process.name, cfg.process.T_true, variant_of(cfg))
123    if out_dir is None:
124        raise FileNotFoundError(f"no results for {cfg.process.name}/{variant_of(cfg)} T={cfg.process.T_true} "
125                                f"under {cfg.paths.output}/{cfg.run_id}")
126    with open(os.path.join(out_dir, "results.json")) as f:
127        blob = json.load(f)
128    rows = blob["aggregate"]
129    if not rows:
130        print("[viz] results.json is empty, nothing to plot")
131        return
132
133    name = f"{blob['sampler']}/{blob['variant']}"               # e.g. ddim/weighted, in every title
134    T_true = blob["config_sweep"]["T_true"]
135    ds, Ks = sorted({r["d"] for r in rows}), sorted({r["K"] for r in rows})
136    models = list(dict.fromkeys(r["model"] for r in rows))      # in order of appearance
137    n_seeds = max(int(r["n_seeds"]) for r in rows)
138
139    made = []
140
141    def attempt(fn, *args):
142        """A failing figure is reported but does not stop the others."""
143        try:
144            made.append(fn(*args))
145        except Exception:
146            print(f"[viz] {fn.__name__} failed:")
147            traceback.print_exc()
148
149    attempt(write_tex, rows, ds, Ks, models, name, T_true, n_seeds, out_dir)
150    attempt(write_png, rows, ds, Ks, models, name, T_true, n_seeds, out_dir)
151    for m in models:
152        attempt(plot_model, best_rows(rows, m), ds, Ks, m,
153                f"{m} (% correct, mean$\\pm$std over {n_seeds} seeds, best anchor budget)", out_dir)
154    for b in sorted({int(r["n_per_mode"]) for r in rows}):
155        b_dir = os.path.join(out_dir, f"anchors_{b}")
156        os.makedirs(b_dir, exist_ok=True)
157        rows_b = [r for r in rows if int(r["n_per_mode"]) == b]
158        for m in models:
159            attempt(plot_model, best_rows(rows_b, m), ds, Ks, m,
160                    f"{m} (% correct, mean$\\pm$std over {n_seeds} seeds, {b} anchors/mode)", b_dir)
161    for p in made:
162        print(f"[viz] wrote {p}")
